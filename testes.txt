# CASOS DE TESTE - SIMULADOR CACHE LRU
# Formato: COMANDO;parametros;resultado_esperado
# Comandos: CAP (capacidade), PUT, GET, CHECK_ORDER, CHECK_SIZE, CLEAR

# ========================================
# TESTE 1: Inserção básica
# ========================================
TESTE;1;Insercao basica em cache vazio
CAP;3
PUT;10;100
CHECK_SIZE;1
CHECK_ORDER;10
PUT;20;200
CHECK_SIZE;2
CHECK_ORDER;20,10
PUT;30;300
CHECK_SIZE;3
CHECK_ORDER;30,20,10
FIM_TESTE

# ========================================
# TESTE 2: Cache HIT move para MRU
# ========================================
TESTE;2;Cache HIT move elemento para MRU
CAP;3
PUT;1;10
PUT;2;20
PUT;3;30
CHECK_ORDER;3,2,1
GET;1;10
CHECK_ORDER;1,3,2
FIM_TESTE

# ========================================
# TESTE 3: Cache MISS retorna -1
